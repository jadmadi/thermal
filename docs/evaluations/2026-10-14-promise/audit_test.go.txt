package audit
import("testing";"os";"path/filepath";"strings")
func TestPromiseAuditCorruptDatabase(t *testing.T){
 h:=t.TempDir();p:=filepath.Join(h,".codex","state_5.sqlite");os.MkdirAll(filepath.Dir(p),0755);os.WriteFile(p,[]byte("this is not sqlite"),0600)
 r:=RunAuditWithOptions(t.TempDir(),h)
 for _,f:=range r.ReadinessFindings{if f.Check=="Tool Databases"{
  t.Logf("%s: %s",f.Status,f.Message)
  low:=strings.ToLower(f.Message)
  if strings.Contains(low,"verified")||strings.Contains(low,"mmap-ready"){t.Errorf("plain text accepted as a verified mmap-ready database: %s",f.Message)}
  if !strings.Contains(low,"filesystem readability")||!strings.Contains(low,"not checked"){t.Errorf("readability limits not disclosed: %s",f.Message)}
 }}
}
